! ==========================================================
! Device: Office-Router (Cisco 2911 ISR)
! Role: Router-on-a-Stick (Inter-VLAN Routing) & DHCP Server
! ==========================================================
hostname Office-Router
no ip domain-lookup

ip dhcp excluded-address 192.168.10.1 192.168.10.10
ip dhcp excluded-address 192.168.10.65 192.168.10.74
ip dhcp excluded-address 192.168.10.129 192.168.10.138

ip dhcp pool VLAN10_IT_POOL
 network 192.168.10.0 255.255.255.192
 default-router 192.168.10.1
 dns-server 8.8.8.8
 exit

ip dhcp pool VLAN20_HR_POOL
 network 192.168.10.64 255.255.255.192
 default-router 192.168.10.65
 dns-server 8.8.8.8
 exit

ip dhcp pool VLAN30_SALES_POOL
 network 192.168.10.128 255.255.255.192
 default-router 192.168.10.129
 dns-server 8.8.8.8
 exit

interface gigabitEthernet 0/0
 description TRUNK_LINK_TO_OFFICE_SWITCH
 no shutdown
 exit

interface gigabitEthernet 0/0.10
 description GATEWAY_VLAN10_IT_ADMIN
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.192
 exit

interface gigabitEthernet 0/0.20
 description GATEWAY_VLAN20_HR_OPERATIONS
 encapsulation dot1Q 20
 ip address 192.168.10.65 255.255.255.192
 exit

interface gigabitEthernet 0/0.30
 description GATEWAY_VLAN30_SALES_SUPPORT
 encapsulation dot1Q 30
 ip address 192.168.10.129 255.255.255.192
 exit

interface gigabitEthernet 0/0.99
 description NATIVE_VLAN99_MANAGEMENT
 encapsulation dot1Q 99 native
 ip address 192.168.10.193 255.255.255.224
 exit

end
